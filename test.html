import requests from bs4 import BeautifulSoup import csv import time import json HEADERS = { "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64)" } def get_all_article_links(query="kesehatan mental", max_page=499): all_links = [] page = 1 while True: print(f"\n🔄 Mengambil halaman ke-{page}...") url = f"https://search.kompas.com/search?q={query}&site_id=all&last_date=all&type=article&page={page}" res = requests.get(url, headers=HEADERS) soup = BeautifulSoup(res.text, "html.parser") # Cari berbagai kemungkinan struktur artikel article_list = soup.find("div", class_=lambda c: c and "articleList" in c) or \ soup.find("ul", class_="article__list") if not article_list: print("❌ Tidak menemukan article_list di halaman ini.") break # Ambil item dari struktur apapun items = article_list.find_all("div", class_="articleItem") if not items: items = article_list.find_all("li", class_="article_list_item") if not items: print("✅ Tidak ada artikel lagi di halaman ini. Stop.") break print(f"📄 Jumlah artikel di halaman {page}: {len(items)}") count_before = len(all_links) for item in items: a_tag = item.find("a") if a_tag and a_tag.get("href"): href = a_tag["href"] if "tv.kompas.com" not in href: all_links.append(href) count_after = len(all_links) print(f"✅ {count_after - count_before} link ditambahkan.") page += 1 time.sleep(1) all_links = list(set(all_links)) # Hapus duplikat return all_links def scrape_article(url): res = requests.get(url, headers=HEADERS) soup = BeautifulSoup(res.text, "html.parser") result = { "url": url, "sumber_berita": "", "jenis_konten": "", "judul": "", "tanggal_publikasi": "", "jam_publikasi": "", "nama_editor": "", "jabatan_editor": "", "konten_berita": "", "tag_berita": "" } # Breadcrumb breadcrumb = soup.find("ul", class_="breadcrumb__wrap") if breadcrumb: items = breadcrumb.find_all("li", class_="breadcrumb__item") if len(items) >= 2: result["sumber_berita"] = items[0].get_text(strip=True) result["jenis_konten"] = items[1].get_text(strip=True) # Judul title_tag = soup.find("h1", class_="read__title") if title_tag: result["judul"] = title_tag.get_text(strip=True) # Bagian header (waktu & editor) header = soup.find("div", class_="read__header") if header: # Ambil waktu publikasi time_div = header.find("div", class_="read__time") if time_div: text = time_div.get_text(strip=True) if "-" in text: time_text = text.split("-")[-1].strip() if "," in time_text: tgl, jam = time_text.split(",") result["tanggal_publikasi"] = tgl.strip() result["jam_publikasi"] = jam.strip() # Ambil informasi editor credit = header.find("div", class_="credit") if credit: # Nama editor name_div = credit.find("div", class_="credit-title-nameEditor") if name_div: result["nama_editor"] = name_div.get_text(strip=True) # Jabatan editor job = credit.find("p") if job: result["jabatan_editor"] = job.get_text(strip=True) # Konten berita content = soup.find("div", class_="read__content") if content: clear = content.find("div", class_="clearfix") if clear: blocks = clear.find_all(["p", "h2"]) paragraphs = [blk.get_text(strip=True) for blk in blocks] result["konten_berita"] = "\n".join(paragraphs) # Tags tags = [] tag_containers = [ soup.find("div", class_="read__tagging mt1 clearfix"), soup.find("div", class_="tag tag--article clearfix"), soup.find("div", class_="tag_article_teaser"), soup.find("ul", class_="tag_article_wrap") ] for container in tag_containers: if container: tag_links = container.find_all("a") for tag in tag_links: tag_text = tag.get_text(strip=True) if tag_text and tag_text not in tags: tags.append(tag_text) result["tag_berita"] = ", ".join(tags) return result # Main execution if __name__ == "__main__": links = get_all_article_links() print(f"🔍 Total link ditemukan: {len(links)}") data = [] for idx, link in enumerate(links): print(f"\n📄 Scraping artikel ke-{idx+1}: {link}") try: article = scrape_article(link) data.append(article) time.sleep(1) except Exception as e: print(f"❌ Error scraping: {e}") if data: # writer.writerows(data) # print("✅ Data disimpan di", nama_file_csv) # Simpan ke JSON nama_file_json = "kesehatan-mental.json" with open(nama_file_json, "w", encoding="utf-8") as f: json.dump(data, f, ensure_ascii=False, indent=4) print("✅ Data disimpan di", nama_file_json) else: print("⚠ Tidak ada data yang berhasil diambil.")
